# Standalone build of decaf-emu's latte-assembler (GFD .gsh writer for Latte
# shader assembly). Only the "assemble" command is built: decaf's "compile"
# path needs AMD's Windows-only shader analyzer.
cmake_minimum_required(VERSION 3.16)
project(latte_assembler_standalone CXX)

set(CMAKE_CXX_STANDARD 17)
set(DECAF_SOURCE_DIR "" CACHE PATH "decaf-emu checkout")
set(D ${DECAF_SOURCE_DIR})

add_subdirectory(${D}/libraries/fmt fmt EXCLUDE_FROM_ALL)

set(GRAMMAR_HEADER ${CMAKE_BINARY_DIR}/grammar.h)
add_custom_command(OUTPUT ${GRAMMAR_HEADER}
                   COMMAND ${CMAKE_COMMAND} -DIN=${D}/tools/latte-assembler/resources/grammar.txt
                           -DOUT=${GRAMMAR_HEADER} -P ${CMAKE_CURRENT_SOURCE_DIR}/grammar.cmake
                   DEPENDS ${D}/tools/latte-assembler/resources/grammar.txt)

# main.cpp and gfd.cpp are patched copies generated by setup.sh.
file(GLOB ASSEMBLER_SOURCES ${D}/tools/latte-assembler/src/*.cpp)
list(FILTER ASSEMBLER_SOURCES EXCLUDE REGEX "/(main|gfd|glsl_compiler)\\.cpp$")
add_executable(latte-assembler ${CMAKE_BINARY_DIR}/main.cpp ${CMAKE_BINARY_DIR}/gfd.cpp assert_shim.cpp ${ASSEMBLER_SOURCES}
               ${D}/src/libgfd/src/gfd_read.cpp ${D}/src/libgfd/src/gfd_write.cpp ${GRAMMAR_HEADER})
target_include_directories(latte-assembler PRIVATE ${CMAKE_BINARY_DIR} ${D}/tools/latte-assembler/src ${D}/src
                           ${D}/src/libgfd ${D}/libraries/excmd/src ${D}/libraries/cpp-peglib
                           ${D}/libraries/cnl/include ${D}/libraries/gsl-lite/include)
target_link_libraries(latte-assembler PRIVATE fmt::fmt)
